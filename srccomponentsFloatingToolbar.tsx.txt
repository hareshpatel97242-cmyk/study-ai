import React from 'react';
import { Camera, CameraOff, Mic, MicOff, Zap, Download, Power } from 'lucide-react';
import { AppState } from '../types';

interface FloatingToolbarProps {
  isCameraOn: boolean;
  onToggleCamera: () => void;
  state: AppState;
  isPowerOn: boolean;
  onTogglePower: () => void;
  onQuickJoke: () => void;
  onQuickPdf: () => void;
}

export const FloatingToolbar: React.FC<FloatingToolbarProps> = ({
  isCameraOn,
  onToggleCamera,
  state,
  isPowerOn,
  onTogglePower,
  onQuickJoke,
  onQuickPdf,
}) => {
  return (
    <footer className="w-full max-w-2xl mx-auto py-5 flex items-center justify-center gap-3 sm:gap-4 z-20">
      <div className="flex items-center gap-2 sm:gap-3 bg-black/60 border border-white/10 backdrop-blur-2xl px-4 sm:px-6 py-3 rounded-full shadow-2xl">
        {/* Camera Toggle */}
        <button 
          onClick={onToggleCamera}
          className={`p-3 rounded-full border transition-all duration-300 active:scale-95 flex items-center justify-center cursor-pointer ${
            isCameraOn 
              ? 'bg-emerald-500 text-black border-transparent shadow-[0_0_15px_rgba(16,185,129,0.4)]' 
              : 'bg-white/5 border-white/10 text-white/50 hover:text-white hover:bg-white/10'
          }`}
          title={isCameraOn ? "કેમેરા બંધ કરો" : "કેમેરા ચાલુ કરો"}
        >
          {isCameraOn ? <Camera size={20} /> : <CameraOff size={20} />}
        </button>
        
        {/* Mic Indicator */}
        <div 
          className={`p-3 rounded-full border border-white/10 bg-white/5 transition-all duration-300 flex items-center justify-center ${
            isPowerOn ? 'opacity-100' : 'opacity-30 pointer-events-none'
          }`}
          title={state === 'listening' ? "માઇક્રોફોન સાંભળી રહ્યો છે" : "માઇક્રોફોન તૈયાર છે"}
        >
          {state === 'listening' ? (
            <Mic size={20} className="text-cyan-400 animate-pulse" />
          ) : (
            <MicOff size={20} className="text-white/30" />
          )}
        </div>

        {/* Quick Power Toggle */}
        <button
          onClick={onTogglePower}
          className={`p-3 rounded-full border transition-all duration-300 active:scale-95 flex items-center justify-center cursor-pointer ${
            isPowerOn 
              ? 'bg-red-500/20 text-red-400 border-red-500/30 hover:bg-red-500/30' 
              : 'bg-cyan-500 text-black border-cyan-400 font-bold hover:bg-cyan-400 shadow-[0_0_15px_rgba(6,182,212,0.4)]'
          }`}
          title={isPowerOn ? "સિસ્ટમ ડિસ્કનેક્ટ કરો" : "લાઇવ કનેક્ટ કરો"}
        >
          <Power size={20} />
        </button>

        {/* Quick PDF Action */}
        <button 
          onClick={onQuickPdf}
          className="p-3 rounded-full border border-white/10 bg-white/5 text-cyan-300 hover:bg-cyan-500/20 hover:border-cyan-500/30 hover:text-cyan-200 transition-all active:scale-95 flex items-center justify-center cursor-pointer"
          title="ઝડપી શોર્ટ નોટ્સ PDF ડાઉનલોડ"
        >
          <Download size={20} />
        </button>

        {/* Quick Joke / Fun Button */}
        <button 
          onClick={onQuickJoke}
          className="p-3 rounded-full border border-white/10 bg-white/5 text-amber-300 hover:bg-amber-500/20 hover:border-amber-500/30 hover:text-amber-200 transition-all active:scale-95 flex items-center justify-center cursor-pointer"
          title="રમુજી જોક સાંભળો"
        >
          <Zap size={20} />
        </button>
      </div>
    </footer>
  );
};
